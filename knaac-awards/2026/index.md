<p>
  Welcome to the third annual knAAC Awards! Short for
  "Keepers and Noteworthies in AAC", the knAAC Awards
  are all about highlighting the wonderful work happening
  in the world of AAC! We want to help celebrate the 
  unsung heroes of AAC who go above and beyond, not for fame or money, but because they see the impact it can make for underserved populations around the world.
</p>
<p>
  All of these people and products have done impressive
  work, and we are grateful for how their 
  contributions have helped AAC users. Our goal is not
  to run a popularity contest, but instead to try
  to bring attention to all of the nominees, and 
  maybe have a little fun like a chilli cook-off
  or voting for our favorite superheroes.
</p>
<p>
  Nominations and voting for the 2026 knAAC Awards
  haven't opened yet. Check back here as the year
  progresses -- once nominees have been gathered and
  voting has concluded, this page will be updated with
  the full list of nominees and that year's highlights
  in each category.
</p>

  var toggle = function(list) {
    list.classList.toggle('expanded');
    var h3 = list.parentNode.querySelector('h3');
    if(h3) {
      if(list.classList.contains('expanded')) {
        h3.classList.add('expanded');
      } else {
        h3.classList.remove('expanded');
      }
    }
  };
  document.querySelectorAll('h3').forEach(function(h3) {
    var list = h3.parentNode.querySelector('.nominees');
    if(list) {
      h3.addEventListener('click', function(e) {
        e.preventDefault();
        toggle(list);
      });
    }
  });
  document.querySelectorAll('.nominees').forEach(function(noms) {
    var list = [];
    noms.querySelectorAll('.nominee').forEach(function(nom) {
      nom.final_rank = parseInt(nom.getAttribute('data-rank'), 10) || 10;
      list.push(nom);
      if(nom.getAttribute('data-rank') == '1') {
        nom.classList.add('winner');
      } else if(nom.getAttribute('data-rank') == '2') {
        nom.classList.add('runner-up');
      }
    });
    list = list.sort(function(a, b) {
      if(a.final_rank != b.final_rank) {
        return a.final_rank - b.final_rank;
      }
      return Math.random() - 0.5;
    });
    list.forEach(function(nom) {
      noms.appendChild(nom);
    });
    noms.addEventListener('click', function(e) {
      var nom = e.target.closest('.nominee');
      if(!noms.classList.contains('expanded') && (!nom || !nom.classList.contains('winner'))) {
        e.preventDefault();
        toggle(noms);
      }
    });
  })
</script>
